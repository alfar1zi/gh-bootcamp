const form = document.querySelector('#daftar');

form.addEventListener('submit', (event) => {
  event.preventDefault();

  const email = form.email.value.trim();

  if (!email.includes('@')) {
    alert('Email tidak valid');
    return;
  }

  console.log('mengirim', email);
});

Tim CodeCraft Academy

CodeCraft Academy dibangun oleh dua anggota yang membagi halaman ini per bagian. Setiap anggota memegang satu bidang saja, mulai dari halaman depan sampai halaman kontak, agar setiap bagian bisa dikerjakan dan diperiksa secara terpisah.

  • Menyusun struktur awal repository dan menangani bagian Beranda pada website CodeCraft Academy. Fokusnya pada tata letak halaman depan, hierarki visual, dan alur navigasi antar bagian. Kontribusinya tercatat pada branch feat/beranda.
